//! Descriptive complete-population reporting from the already accepted auditor.
use serde_json::{json,Value};
use std::{collections::BTreeMap,env,fs};
fn main(){
 let a:Vec<_>=env::args().collect();assert_eq!(a.len(),3);assert!(!std::path::Path::new(&a[2]).exists());let v:Value=serde_json::from_slice(&fs::read(&a[1]).unwrap()).unwrap();let calls=v["all60_call_results"].as_array().unwrap();assert_eq!(calls.len(),60);
 let(mut row_failures,mut col_disagree_calls,mut known_disagree_calls,mut bad_class_legs,mut bad_source_legs,mut bad_y_legs,mut source_ratio_bad,mut original_bad)=(0u64,0u64,0u64,0u64,0u64,0u64,0u64,0u64);
 let mut groups=BTreeMap::<(u64,u64),Vec<Value>>::new();let mut by_fb=BTreeMap::<u64,Value>::new();
 for c in calls{
  let side=c["identity"][0].as_u64().unwrap();let setting=c["identity"][2].as_u64().unwrap();let fb=[30,36,48,64][setting as usize];
  let r=c["available_relation_character_failures"].as_u64().unwrap();let cols=c["candidate_column_disagreements"].as_u64().unwrap();let known=c["known_log_disagreements"].as_u64().unwrap();
  let d=c["all_three_descent_checks"].as_array().unwrap();assert_eq!(d.len(),3);let bad_class=d.iter().filter(|v|v["actual_class_decomposition_equal"]!=true).count()as u64;let bad_source=d.iter().filter(|v|v["source_decomposition_equal"]!=true).count()as u64;let bad_y=d.iter().filter(|v|v["candidate_y_agrees"]!=true).count()as u64;
  let sr=c["direct_actual_target_ratio_solution"]!=true;let ob=c["original_candidate_actual_solution"]!=true;row_failures+=r;col_disagree_calls+=u64::from(cols!=0);known_disagree_calls+=u64::from(known!=0);bad_class_legs+=bad_class;bad_source_legs+=bad_source;bad_y_legs+=bad_y;source_ratio_bad+=u64::from(sr);original_bad+=u64::from(ob);
  let entry=by_fb.entry(fb).or_insert(json!({"calls":0,"original_bad":0,"source_ratio_bad":0,"relation_character_bad":0,"column_disagree_calls":0,"known_disagree_calls":0,"bad_class_legs":0,"bad_source_legs":0,"bad_y_legs":0}));
  for(name,add)in[("calls",1),("original_bad",u64::from(ob)),("source_ratio_bad",u64::from(sr)),("relation_character_bad",r),("column_disagree_calls",u64::from(cols!=0)),("known_disagree_calls",u64::from(known!=0)),("bad_class_legs",bad_class),("bad_source_legs",bad_source),("bad_y_legs",bad_y)]{entry[name]=json!(entry[name].as_u64().unwrap()+add);}
  let rank=c["rank_certificates"].as_array().unwrap().iter().find(|r|r["family"]=="all_available"&&r["restriction"]=="full").unwrap();
  groups.entry((side,fb)).or_default().push(json!({"identity":c["identity"],"rank":rank["rank"],"nullity":rank["nullity"],"point_witnesses":c["point_witnesses"],"eta_character":c["eta_character"],"generator_character":c["generator_character"],"anchor_column":c["anchor_column"],"multiplier":c["multiplier"],"candidate_column_disagreements":cols,"known_log_disagreements":known,"bad_class_legs":bad_class,"bad_source_legs":bad_source,"bad_y_legs":bad_y,"source_ratio":c["source_character_ratio"],"source_ratio_direct_target_solution":!sr,"original_candidate_actual_solution":!ob}));
 }
 let table:Vec<_>=groups.into_iter().map(|((side,fb),rows)|{assert_eq!(rows.len(),5);json!({"side":side,"factor_base":fb,"all_five_public_observations":rows,"rank_all_five_equal":rows.iter().all(|r|r["rank"]==rows[0]["rank"]),"nullity_all_five_equal":rows.iter().all(|r|r["nullity"]==rows[0]["nullity"]),"candidate_column_disagreements_all_five_equal":rows.iter().all(|r|r["candidate_column_disagreements"]==rows[0]["candidate_column_disagreements"]),"known_log_disagreements_all_five_equal":rows.iter().all(|r|r["known_log_disagreements"]==rows[0]["known_log_disagreements"])})}).collect();
 let out=json!({"protocol":"8e7e1c","source":"whole accepted direct-source auditor verdict, all60calls retained","whole_calls":60,"checked_class_witnesses":v["whole_checked_class_witnesses"],"checked_available_relations":v["whole_checked_available_relations"],"relation_character_failures":row_failures,"candidate_column_disagreement_calls":col_disagree_calls,"known_log_disagreement_calls":known_disagree_calls,"canonical_class_decomposition_failed_legs":bad_class_legs,"source_character_decomposition_failed_legs":bad_source_legs,"candidate_y_disagreement_legs":bad_y_legs,"source_ratio_direct_target_failure_calls":source_ratio_bad,"original_candidate_failure_calls":original_bad,"by_factor_base":by_fb.into_iter().map(|(fb,v)|(fb.to_string(),v)).collect::<serde_json::Map<String,Value>>(),"all_twelve_groups":table,"scope":"descriptive finite complete-population projection only; original full records, scalar/group checks, tables and witnesses retained; no causal/global-character/cost theorem"});
 fs::write(&a[2],serde_json::to_vec_pretty(&out).unwrap()).unwrap();
}
